import React, { useState } from "react";
import { View, Text, TextInput, ScrollView, StyleSheet, Switch, Alert, Image, FlatList, Modal, TouchableOpacity } from "react-native";
import { createBottomTabNavigator } from "@react-navigation/bottom-tabs";
import { NavigationContainer } from "@react-navigation/native";
import RNPickerSelect from "react-native-picker-select";
import * as ImagePicker from "expo-image-picker";
import { FloatingAction } from "react-native-floating-action";

// ----- Lamu County Data -----
const subCounties = [
  { label: "Lamu East", value: "Lamu East" },
  { label: "Lamu West", value: "Lamu West" },
];

const wards = {
  "Lamu East": [
    { label: "Hindi", value: "Hindi" },
    { label: "Faza", value: "Faza" },
  ],
  "Lamu West": [
    { label: "Shela", value: "Shela" },
    { label: "Mkomani", value: "Mkomani" },
  ],
};

const violenceTypes = [
  { label: "Physical", value: "Physical" },
  { label: "Psychological", value: "Psychological" },
  { label: "Sexual", value: "Sexual" },
];

// ----- Victim Screen -----
function VictimScreen() {
  const [name, setName] = useState("");
  const [age, setAge] = useState("");
  const [subCounty, setSubCounty] = useState("");
  const [ward, setWard] = useState("");
  const [village, setVillage] = useState("");
  const [type, setType] = useState("");
  const [description, setDescription] = useState("");
  const [image, setImage] = useState(null);
  const [anonymous, setAnonymous] = useState(false);

  const pickImage = async () => {
    let result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.All,
      allowsEditing: true,
      quality: 1,
    });
    if (!result.canceled) setImage(result.assets[0].uri);
  };

  const submitReport = () => {
    if (!type || !description || !subCounty || !ward || !village) {
      Alert.alert("Error", "Please fill all required fields");
      return;
    }

    const reportData = {
      name: anonymous ? "Anonymous" : name,
      age,
      location: { county: "Lamu", subCounty, ward, village },
      type,
      description,
      image,
      date: new Date().toLocaleString(),
    };

    console.log("Report Submitted:", reportData);

    Alert.alert(
      "Thank You",
      "Your report has been successfully submitted. We appreciate your courage in reporting this incident."
    );

    // Reset form
    setName("");
    setAge("");
    setSubCounty("");
    setWard("");
    setVillage("");
    setType("");
    setDescription("");
    setImage(null);
    setAnonymous(false);
  };

  const fabActions = [
    { text: "Submit Report", name: "bt_submit", position: 1, color: "#007bff" },
    { text: "Upload Evidence", name: "bt_upload", position: 2, color: "#28a745" },
  ];

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.welcomeText}>
        Welcome to the GBV Reporting App. Your safety and voice matter. Please fill out the form below to report any incidents.
      </Text>

      <Switch
        value={anonymous}
        onValueChange={setAnonymous}
        style={{ marginBottom: 10 }}
      />
      <Text>{anonymous ? "Reporting as Anonymous" : "Reporting with Name"}</Text>

      {!anonymous && (
        <>
          <TextInput style={styles.input} placeholder="Full Name" value={name} onChangeText={setName} />
          <TextInput style={styles.input} placeholder="Age" keyboardType="numeric" value={age} onChangeText={setAge} />
        </>
      )}

      <RNPickerSelect
        onValueChange={(value) => { setSubCounty(value); setWard(""); }}
        items={subCounties}
        placeholder={{ label: "Select Sub-County", value: "" }}
        value={subCounty}
        style={pickerSelectStyles}
      />

      {subCounty ? (
        <RNPickerSelect
          onValueChange={setWard}
          items={wards[subCounty]}
          placeholder={{ label: "Select Ward", value: "" }}
          value={ward}
          style={pickerSelectStyles}
        />
      ) : null}

      <TextInput style={styles.input} placeholder="Village" value={village} onChangeText={setVillage} />
      <TextInput style={styles.input} placeholder="Type of Violence" value={type} onChangeText={setType} />
      <TextInput style={[styles.input, { height: 100 }]} placeholder="Description" multiline value={description} onChangeText={setDescription} />

      {image && <Image source={{ uri: image }} style={{ width: 200, height: 200, marginVertical: 10 }} />}

      <FloatingAction
        actions={fabActions}
        onPressItem={(name) => {
          if (name === "bt_submit") submitReport();
          if (name === "bt_upload") pickImage();
        }}
      />
    </ScrollView>
  );
}

// ----- Admin Screen -----
function AdminScreen() {
  const [reports, setReports] = useState([]);
  const [filteredReports, setFilteredReports] = useState([]);
  const [modalVisible, setModalVisible] = useState(false);
  const [filterSubCounty, setFilterSubCounty] = useState("");
  const [filterWard, setFilterWard] = useState("");
  const [filterType, setFilterType] = useState("");

  const applyFilter = () => {
    const results = reports.filter((r) => {
      return (
        (!filterSubCounty || r.location.subCounty === filterSubCounty) &&
        (!filterWard || r.location.ward === filterWard) &&
        (!filterType || r.type === filterType)
      );
    });
    setFilteredReports(results);
    setModalVisible(false);
  };

  const resetFilter = () => {
    setFilterSubCounty("");
    setFilterWard("");
    setFilterType("");
    setFilteredReports(reports);
    setModalVisible(false);
  };

  const fabActions = [{ text: "Filter Reports", name: "bt_filter", position: 1, color: "#ffc107" }];

  return (
    <View style={styles.container}>
      <Text style={styles.header}>Admin Dashboard</Text>

      <FlatList
        data={filteredReports.length ? filteredReports : reports}
        keyExtractor={(item, index) => index.toString()}
        renderItem={({ item }) => (
          <View style={styles.reportCard}>
            <Text style={{ fontWeight: "bold" }}>Reporter: {item.name}</Text>
            <Text>Age: {item.age}</Text>
            <Text>Location: {item.location.subCounty}, {item.location.ward}, {item.location.village}</Text>
            <Text>Type: {item.type}</Text>
            <Text>Description: {item.description}</Text>
            <Text>Date: {item.date}</Text>
            {item.image && <Image source={{ uri: item.image }} style={{ width: 200, height: 200, marginTop: 5 }} />}
          </View>
        )}
      />

      <Modal visible={modalVisible} animationType="slide" transparent={true}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={{ fontWeight: "bold", fontSize: 18, marginBottom: 10 }}>Filter Reports</Text>

            <RNPickerSelect
              onValueChange={(value) => { setFilterSubCounty(value); setFilterWard(""); }}
              items={subCounties}
              placeholder={{ label: "Select Sub-County", value: "" }}
              value={filterSubCounty}
              style={pickerSelectStyles}
            />

            {filterSubCounty ? (
              <RNPickerSelect
                onValueChange={setFilterWard}
                items={wards[filterSubCounty]}
                placeholder={{ label: "Select Ward", value: "" }}
                value={filterWard}
                style={pickerSelectStyles}
              />
            ) : null}

            <RNPickerSelect
              onValueChange={setFilterType}
              items={violenceTypes}
              placeholder={{ label: "Select Type of Violence", value: "" }}
              value={filterType}
              style={pickerSelectStyles}
            />

            <View style={{ flexDirection: "row", justifyContent: "space-between", marginTop: 15 }}>
              <TouchableOpacity style={styles.modalButton} onPress={applyFilter}>
                <Text style={{ color: "#fff", fontWeight: "bold" }}>Apply</Text>
              </TouchableOpacity>
              <TouchableOpacity style={[styles.modalButton, { backgroundColor: "#6c757d" }]} onPress={resetFilter}>
                <Text style={{ color: "#fff", fontWeight: "bold" }}>Reset</Text>
              </TouchableOpacity>
            </View>

            <TouchableOpacity style={{ marginTop: 10, alignSelf: "center" }} onPress={() => setModalVisible(false)}>
              <Text style={{ color: "#007bff" }}>Close</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>

      <FloatingAction
        actions={fabActions}
        onPressItem={(name) => {
          if (name === "bt_filter") setModalVisible(true);
        }}
      />
    </View>
  );
}

// ----- Navigation -----
const Tab = createBottomTabNavigator();

export default function App() {
  return (
    <NavigationContainer>
      <Tab.Navigator>
        <Tab.Screen name="Victim" component={VictimScreen} />
        <Tab.Screen name="Admin" component={AdminScreen} />
      </Tab.Navigator>
    </NavigationContainer>
  );
}

// ----- Styles -----
const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, backgroundColor: "#f2f2f2" },
  header: { fontSize: 24, fontWeight: "bold", marginBottom: 20, textAlign: "center" },
  welcomeText: { fontSize: 16, fontWeight: "500", marginBottom: 20, textAlign: "center", color: "#333" },
  input: { backgroundColor: "#fff", padding: 10, marginVertical: 5, borderRadius: 8 },
  reportCard: { backgroundColor: "#fff", padding: 15, borderRadius: 8, marginVertical: 5 },
  modalOverlay: { flex: 1, justifyContent: "center", backgroundColor: "rgba(0,0,0,0.5)" },
  modalContent: { backgroundColor: "#fff", margin: 20, padding: 20, borderRadius: 10 },
  modalButton: { backgroundColor: "#007bff", padding: 10, borderRadius: 8, flex: 1, alignItems: "center", marginHorizontal: 5 },
});

const pickerSelectStyles = StyleSheet.create({
  inputIOS: { backgroundColor: "#fff", padding: 10, marginVertical: 5, borderRadius: 8 },
  inputAndroid: { backgroundColor: "#fff", padding: 10, marginVertical: 5, borderRadius: 8 },
});
